def _processos():
    """{pid: (minutos de vida, linha de comando)} dos python/wscript/cmd do PC1; {} se a consulta falhar."""

**ENTRAR:**
```python
def _processos():
    """{pid: (minutos de vida, linha de comando)} dos python/wscript/cmd do PC1; {} se a consulta falhar."""
    try:
        # WMIC em vez de PowerShell (PowerShell intermitentemente trava 90s; WMIC é instantâneo)
        r = subprocess.run(
            ["wmic", "process", "where", 
             "name='python.exe' or name='pythonw.exe' or name='wscript.exe' or name='cmd.exe'",
             "get", "ProcessId,CreationDate,CommandLine", "/format:csv"],
            capture_output=True, text=True, timeout=30,
            encoding="utf-8", errors="replace", creationflags=_SEM_JANELA)
    except Exception as e:  # noqa: BLE001
        log(f"[!] lista de processos falhou: {e}")
        return {}
    out = {}
    for linha in (r.stdout or "").splitlines()[1:]:  # pula cabeçalho CSV
        if not linha.strip():
            continue
        partes = linha.split(",")
        if len(partes) >= 4 and partes[2].strip().isdigit():
            pid = int(partes[2])
            cmd = ",".join(partes[3:]).strip()  # CommandLine pode ter vírgulas
            # CreationDate formato: 20261024213253.292321-180 → extrair minutos
            try:
                import datetime
                data_str = partes[1].split(".")[0]  # pega só AAAAMMDDHHMMSS
                criacao = datetime.datetime.strptime(data_str, "%Y%m%d%H%M%S")
                minutos = int((datetime.datetime.now() - criacao).total_seconds() / 60)
            except (ValueError, IndexError):
                minutos = 0
            out[pid] = (minutos, cmd)
    return out
```
